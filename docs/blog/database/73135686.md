/* * 作用:连接数据库 *
 */
function connect($db_host, $db_user, $db_pass, $db_name, $db_charset)
{
  @mysql_connect($db_host, $db_user, $db_pass);
  mysql_select_db($db_name);
  mysql_set_charset($db_charset);
}
/* * 作用:遍历结果集返回一个数组 *
 */
function getAll($sql)
{
  $res = mysql_query($sql);
  $rows = [];
  while ($row = mysql_fetch_assoc($res)) {
    $rows[] = $row;
  }
  return $rows;
}
/* * * 作用:关闭数据库资源
 */
function close()
{
  mysql_close();
