namespace app\index\controller;

  $user = new UserModel;
  $user-&gt;data([
    'name' =&gt; 'thinkphp',
    'email' =&gt; 'thinkphp@qq.com'
  ]);
  $count = $user-&gt;save();

  $user-&gt;email = 'thinkphp@qq.com';
