 * Render an exception into an HTTP response. * * @param  \Illuminate\Http\Request  $request * @param  \Exception  $exception * @return \Illuminate\Http\Response
 */
public function render($request, Exception $exception)
{
  if (config('app.debug')) {
    $replacements = [];
    $error = $this-&gt;convertExceptionToResponse($exception);
    $statusCode = $error-&gt;getStatusCode($exception);
    if (! $message = $exception-&gt;getMessage()) {
      $message = sprintf('%d %s', $statusCode, Response::$statusTexts[$statusCode]);
    }
    $replacements = [
      'errcode' =&gt; $statusCode,
      'errdesc' =&gt; $message,
    ];
    if ($exception instanceof MessageBagErrors &amp;&amp; $exception-&gt;hasErrors()) {
      $replacements['errors'] = $exception-&gt;getErrors();
    }
    if ($code = $exception-&gt;getCode()) {
      $replacements['code'] = $code;
    }
    if (!config('app.debug')) {
      $replacements['debug'] = [
        'line' =&gt; $exception-&gt;getLine(),
        'file' =&gt; $exception-&gt;getFile(),
        'class' =&gt; get_class($exception),
        'trace' =&gt; explode("\n", $exception-&gt;getTraceAsString()),
      ];
    }
    return response($replacements, $statusCode);
  }
  return parent::render($request, $exception);
}</code></pre></div>
