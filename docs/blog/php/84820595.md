<p>1、使用QQ邮箱</p>
<div class="language-php vp-adaptive-theme"><button title="Copy Code" class="copy"></button><span class="lang">php</span><pre class="vp-code" tabindex="0" v-pre><code>$config['protocol'] = 'smtp';
$config['smtp_host'] = 'smtp.qq.com';
$config['smtp_user'] = '675667823@qq.com';
$config['smtp_pass'] = '************';
//授权码，不是QQ邮箱密码
$config['smtp_port'] = 465;
$config['smtp_crypto'] = 'ssl';
$config['charset'] = 'utf-8';
$config['wordwrap'] = TRUE;
$config['mailtype'] = 'html';
$config['crlf'] = "\r\n";
$config['newline'] = "\r\n";
//必须是双引号
$this-&gt;load-&gt;library('email');
$this-&gt;email-&gt;initialize($config);
$this-&gt;email-&gt;from('675667823@qq.com', 'TEST');
//发送方
$this-&gt;email-&gt;to('leroiliu@163.com');
//接收方
$this-&gt;email-&gt;subject('SUBJECT');
//主题
$this-&gt;email-&gt;message('MESSAGE');
//邮件内容
$this-&gt;email-&gt;send();
//发送</code></pre></div>
