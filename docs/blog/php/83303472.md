<h2>5、使用<strong>DB façade</strong>和<strong>查询构造器</strong>操作数据库</h2>
<div class="language-php vp-adaptive-theme"><button title="Copy Code" class="copy"></button><span class="lang">php</span><pre class="vp-code" tabindex="0" v-pre><code>&lt;?php
namespace App\Http\Controllers;
use Illuminate\Support\Facades\DB;
class StudentController extends Controller
{
  // public function index(){
    //使用DB façade操作数据库
    $bool = DB::insert('insert into student(name,age) values(?,?)', ['Leroi', 18]);
    var_dump($bool);
    $num = DB::update('update student
    set age = ?
    where name = ?',[20,'Leroi']);

    dd($students);

    $bool = DB::table('student')-&gt;insert(['name' =&gt; 'Li', 'age' =&gt; 18]);

    $id = DB::table('student')-&gt;insertGetId(['name' =&gt; 'LiL', 'age' =&gt; 28]);
    var_dump($id);
    //插入多条数据
    $bool = DB::table('student')-&gt;insert([['name' =&gt; 'Lpi', 'age' =&gt; 16], ['name' =&gt; 'Lia', 'age' =&gt; 108]]);
    var_dump($bool);

    $num = DB::table('student')-&gt;where('id', 2)-&gt;update(['age' =&gt; 50]);

    //删除
    $num = DB::table('student')-&gt;where('id', '&gt;=', 4)-&gt;delete();

    /////////多注释几个以防万一/////////$num = DB::table('student')-&gt;truncate();
    var_dump($num);
    //获取表的所有数据
    $students = DB::table('student')-&gt;get();
    dd($students);
    $students = DB::table('student')-&gt;orderBy('id', 'desc')-&gt;where('id', '&gt;=', 2)-&gt;get();
    dd($students);
    //获取结果信息第一条
    $students = DB::table('student')-&gt;orderBy('id', 'desc')-&gt;first();
    dd($students);
    //多条件查询
    $students = DB::table('student')-&gt;where('id', '&gt;', 2)-&gt;where('id', '&lt;', 4)-&gt;get();
    dd($students);
    $students = DB::table('student')-&gt;whereRaw('id&gt;= ? and id &lt; ?', [2, 4])-&gt;get();
    dd($students);
    //pluck
    $names = DB::table('student')-&gt;pluck('name');
    dd($names);
    //指定键名
    $names = DB::table('student')-&gt;pluck('name', 'id');
    dd($names);
    //select
    $students = DB::table('student')-&gt;select('id', 'name', 'age')-&gt;get();
    dd($students);
    //chunk
    DB::table('student')-&gt;orderBy('id', 'desc')-&gt;chunk(2, function ($students) {
      var_dump($students);
      if (condition) {
        return false;
      }
    });
    //聚合函数

<h2>6、使用ORM进行数据库操作</h2>
<div class="language-php vp-adaptive-theme"><button title="Copy Code" class="copy"></button><span class="lang">php</span><pre class="vp-code" tabindex="0" v-pre><code>//模型 &lt;?php
namespace App;
use Illuminate\Database\Eloquent\Model;
class Student extends Model
{
  //指定表名
  protected $table = 'student';
  //指定id
  protected $primaryKey = 'id';
  //是否自动维护时间戳
  public $timestamps = true;
  //指定可以批量赋值的字段
  public $fillable = ['name', 'age'];
  //指定字段不可以批量赋值
  public $guarded = [];
  public function getDateFormat()
  {
    return time();
  }
}</code></pre></div>
<div class="language-php vp-adaptive-theme"><button title="Copy Code" class="copy"></button><span class="lang">php</span><pre class="vp-code" tabindex="0" v-pre><code>//操作 &lt;?php
namespace App\Http\Controllers;
use Illuminate\Support\Facades\DB;
use App\Student;
class StudentController extends Controller
{
  public function index()
  {
    //使用ORM进行数据库操作 //全部查找
    $students = Student::all();
    //根据主键查找
    $students = Student::find(1);
    //根据主键查找，没有则抛出异常
    $students = Student::findOrFail(1);
    //使用查询构造器查询
    $students = Student::get();
    $students = Student::where('id', '&gt;', 1)-&gt;orderBy('age', 'desc')-&gt;first();
    Student::orderBy('id', 'desc')-&gt;chunk(2, function ($students) {
      var_dump($students);
    });
    $num = Student::count();
    //使用模型新增数据
    $student = new Student();
    $student-&gt;name = '哈哈';
    $student-&gt;age = 108;
    $bool = $student-&gt;save();
    dd($bool);
    //使用模型Create方法新增数据
    $student = Student::create(['name' =&gt; 'ico', 'age' =&gt; 18]);
    //firstOrCreate以属性查找用户，若没有则新增
    $student = Student::firstOrCreate(['name' =&gt; 'ipco']);
    //firstOrNew以属性查找用户，若没有则返回新的实例，若要保存可保存实例
    $student = Student::firstOrNew(['name' =&gt; 'ipcoooo']);
    $bool = $student-&gt;save();
    dd($bool);
    //通过模型更新数据
    $student = Student::find(1);
    $student-&gt;name = 'kitty';
    $bool = $student-&gt;save();
    var_dump($bool);
    $num = Student::where('id', '&gt;', 2)-&gt;update(['age' =&gt; 41]);
    var_dump($num);
    //通过模型删除
    $student = Student::find(1);
    $bool = $student-&gt;delete();
    //通过主键删除
    $num = Student::destory(1);
    $num = Student::destory(2, 3);
    $num = Student::destory([2, 3]);
    $num = Student::where('id', '&gt;', 1)-&gt;delete();
  }
}</code></pre></div>
