## 监控命令及事件：

示例代码如下：

~~~php
using System;
using System.IO;
using System.Collections.Generic;
using System.Linq;
using System.Text;
using System.Threading.Tasks;
using Newtonsoft.Json;
using Autodesk.AutoCAD.Runtime;
using Autodesk.AutoCAD.ApplicationServices;
using Autodesk.AutoCAD.DatabaseServices;
using Autodesk.AutoCAD.EditorInput;
namespace Demo1
{
  public class Class1
  {
    [CommandMethod("AddAppEvent")]
    public void AddAppEvent()
    {
      Document acDoc = Application.DocumentManager.MdiActiveDocument;
      Application.SystemVariableChanged + = new Autodesk.AutoCAD.ApplicationServices.SystemVariableChangedEventHandler(appSysVarChanged);
      Application.BeginDoubleClick + = new BeginDoubleClickEventHandler(beginDoubleClick);
      acDoc.CommandWillStart + = new CommandEventHandler(currentDoc_CommandWillStart);
      acDoc.CommandCancelled + = new CommandEventHandler(currentDoc_CommandWillStart);
      acDoc.CommandEnded + = new CommandEventHandler(currentDoc_CommandWillStart);
      acDoc.CommandFailed + = new CommandEventHandler(currentDoc_CommandWillStart);
    }
    [CommandMethod("RemoveAppEvent")]
    public void RemoveAppEvent()
    {
      Application.SystemVariableChanged - = new Autodesk.AutoCAD.ApplicationServices.SystemVariableChangedEventHandler(appSysVarChanged);
      Application.BeginDoubleClick - = new BeginDoubleClickEventHandler(beginDoubleClick);
    }
    public void appSysVarChanged(object senderObj, Autodesk.AutoCAD.ApplicationServices.SystemVariableChangedEventArgs sysVarChEvtArgs)
    {
      object oVal = Application.GetSystemVariable(sysVarChEvtArgs.Name);
      // Display a message box with the system variable name and the new value
      Application.ShowAlertDialog(sysVarChEvtArgs.Name + " was changed." + "\nNew value: " + oVal.ToString());
    }
    static void beginDoubleClick(object sender, BeginDoubleClickEventArgs e)
    {
      Document acDoc = Application.DocumentManager.MdiActiveDocument;
      string strDWGName = acDoc.Name;
      Application.ShowAlertDialog(strDWGName);
    }
    void currentDoc_CommandWillStart(object sender, CommandEventArgs e)
    {
      //Application.ShowAlertDialog(e.GlobalCommandName);
      Document acDoc = Application.DocumentManager.MdiActiveDocument;
      string strDWGName = acDoc.Name;
      Application.ShowAlertDialog(strDWGName);
    }
  }
}
~~~
